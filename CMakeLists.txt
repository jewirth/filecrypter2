cmake_minimum_required(VERSION 3.20)
project(filecrypter2 VERSION 1.0.0 LANGUAGES C)
find_package(OpenSSL 3.0 REQUIRED COMPONENTS Crypto)
add_executable(filecrypter2 filecrypter.c)
target_compile_features(filecrypter2 PRIVATE c_std_11)
target_link_libraries(filecrypter2 PRIVATE OpenSSL::Crypto)
if(MSVC)
  target_compile_options(filecrypter2 PRIVATE /W4)
  target_compile_definitions(filecrypter2 PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(filecrypter2 PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()
install(TARGETS filecrypter2 RUNTIME DESTINATION bin)

include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME cli_security COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:filecrypter2>)
endif()
